import { HistoryFlow, PatientNode, FamilyNode, DiagnosisNode, MedicationNode, TherapyNode } from '@behavioral-health/flow';

export function PatientCanvas() {
  return (
    <HistoryFlow
      nodes={patientNodes}
      edges={patientEdges}
      nodeTypes={{ patient: PatientNode, family: FamilyNode, diagnosis: DiagnosisNode, medication: MedicationNode, therapy: TherapyNode }}
      fitView
      snapToGrid
      a11yTreeAnnouncement
    />
  );
}
<PatientNode />

Root origin node. Exposes 4 omnidirectional connector handles. Displays core identity, MRN, and care episode status.

<MedicationNode />

Pharmacological node with dose, frequency, titration timeline, and prescription discontinuation indicators.

<AssessmentNode />

Metric container for PHQ-9, GAD-7, and Columbia C-SSRS scoring. Shows numeric deltas and severity classifications.
